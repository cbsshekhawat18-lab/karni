"""Pre-0.2 code: every deprecated soft-delete spelling the 0.2 changelog lists."""

from django.db.models import Manager

from karni.managers import BaseQuerySet, SoftDeleteManagerMixin

from shop.models import Event, Venue


class EventQuerySet(BaseQuerySet):
    def upcoming(self):
        return self


class EventManager(SoftDeleteManagerMixin, Manager.from_queryset(EventQuerySet)):
    pass


def everything(venue):
    rows = Event.all_objects.filter(venue=venue)
    gone = Event.objects.deleted_only()
    all_rows = Venue.objects.all_with_deleted()
    related = venue.event_set.all_with_deleted()
    return rows, gone, all_rows, related
